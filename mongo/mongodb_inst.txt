//聚合框架查詢SELECT empno, ename, salary, salary*12 AS '年薪' FROM emp WHERE deptno = 20
// https://www.mongodb.com/zh-cn/docs/manual/reference/mql/aggregation-stages/
db.emp.aggregate([
    // step1：找尋條件(WHERE deptno = 20)
    {
        $match: {
            "deptno": 20
        }
    },
    // step2：投影、選取欄位和計算 (SELECT empno, ename, salary, salary*12 AS '年薪')
    {
        $project: {
            "empno": 1,
            "ename": 1,
            "salary": 1,
            // 計算新欄位 (年薪 = salary * 12)
            "年薪": { 
                $multiply: ["$salary", 12] 
            },
            "_id": 0 
        }
    }
])
